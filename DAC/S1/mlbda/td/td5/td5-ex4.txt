CREATE TABLE Realisateur (
    nom VARCHAR2(30) PRIMARY KEY,
    age NUMBER(3)
);

CREATE TABLE Producteur (
    nom VARCHAR2(30) PRIMARY KEY,
    budget NUMBER(10)
);

CREATE TABLE Film (
    numFilm NUMBER(10) PRIMARY KEY,
    titre VARCHAR2(50),
    realisateur VARCHAR2(30) REFERENCES Realisateur(nom),
    producteur VARCHAR2(30) REFERENCES Producteur(nom)
);

CREATE TABLE Programme (
    salle VARCHAR(3),
    jour VARCHAR(10),
    numFilm NUMBER(10) REFERENCES Film(numFilm),
    nbEntrees NUMBER(10),
    PRIMARY KEY(salle, jour)
);
