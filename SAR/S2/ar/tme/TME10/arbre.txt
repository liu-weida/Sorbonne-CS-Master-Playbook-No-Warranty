#define N 6           /* Number of processes */ 
#define NB_V_MAX 3    /* maximum degree of a node */ 
byte message_count = 0;
byte finished_count = 0;
mtype = {calcul};

chan channel_IN[N] = [N] of {mtype, byte, byte}; 
   /*  All messages to process i are received on channel_IN[i] 
       a message contains information <type, sender, value>   */
   
/**************************************************************/

inline Simulateur() {
   if
      :: (id == 0) ->
	     minloc = 3; nb_voisins = 2;
	     voisins[0] = 1; voisins[1] = 2;
      :: (id == 1) ->
	     minloc = 11; nb_voisins = 3;
	     voisins[0] = 0; voisins[1] = 3; voisins[2] = 4;
      :: (id == 2) ->
	     minloc = 8; nb_voisins = 2;
	     voisins[0] = 0; voisins[1] = 5;
      :: (id == 3) ->
	     minloc = 14; nb_voisins = 1;
	     voisins[0] = 1;
      :: (id == 4) ->
	     minloc = 5; nb_voisins = 1;
	     voisins[0] = 1;
      :: (id == 5) ->
	     minloc = 17; nb_voisins = 1;
	     voisins[0] = 2;
   fi;
   mincal = minloc	  
}

/**************************************************************/
inline Test_Emission (q) {
   /* determine si l'on peut emettre et identifie le destinataire q */
   int nbvoi = nb_voisins;
   int nbnonrecu = 0;
   do
      ::nbvoi>0 ->
         if
            ::recu[voisins[nbvoi-1]] == 1 -> ;
            ::recu[voisins[nbvoi-1]] == 0 -> 
               nbnonrecu ++;
               q = voisins[nbvoi-1];
         fi;
         nbvoi --;
      ::nbvoi==0 ->
         break;
   od;
   if
      ::nbnonrecu == 1 ->
         emission = 1;
         printf("envoi (%d->%d)\n",id,proc);
         father = proc;
         channel_IN[proc] ! calcul,id,mincal;
         atomic{message_count++};
      ::nbnonrecu >1 ->
      ::nbnonrecu == 0 ->
         decision = 1;
         emission = 1;
   fi;
}

inline Test_Decision () {
   /* determine si l'on peut decider */
   int nbvoi = nb_voisins;
   bool toutemi = 1;
   do
      ::if
         ::nbvoi>0 ->
            if
               ::recu[voisins[nbvoi-1]] == 1 -> ;
               ::recu[voisins[nbvoi-1]] == 0 -> 
                  toutemi = 0;
                  break;
            fi;
            nbvoi--;
        fi;    
      ::if
         ::nbvoi==0 ->
         break;
      fi;
   od;
   if
      ::toutemi == 1 -> decision = 1;
      ::toutemi == 0 ->;
   fi;
}
inline EnvoiMinVal(){
   int nbvoi = nb_voisins;
   do
      ::if
         ::nbvoi>0 ->
            if
               ::voisins[nbvoi-1] == father ->;
               ::voisins[nbvoi-1] != father ->
                  printf("renvoi (%d->%d)\n",id,voisins[nbvoi-1]);
                  channel_IN[voisins[nbvoi-1]] ! calcul,id,mincal;
                  atomic{message_count++};
            fi;
            nbvoi--;
        fi;    
      ::if
         ::nbvoi==0 ->
         break;
      fi;
   od;
   deja_emis = 1;
}

/**************************************************************/

proctype node( byte id ) {
    
byte nb_voisins;
byte voisins[NB_V_MAX];

chan canal_IN = channel_IN[id];
xr canal_IN;            /* only id reads on this channel */

mtype type;
byte i, nb_recus,father = 255;
byte sender, receiver, minloc, mincal,valrec;

/* tableau initialise a 0 ; recu[i] = 1 si on a recu de i */
byte recu[N];  
bool emission = 0, wait = 0, decision = 0, deja_emis = 0;
Simulateur();
/* Each process executes a finite loop */
   do
      :: /* test emission et traitement correspondant */
         emission == 0 && decision == 0->
            int proc ;
            Test_Emission(proc);         
      :: /* test reception et traitement correspondant */
         canal_IN ? type, sender, valrec ->
            printf("recu (%d->%d)\n",sender,id);
            recu[sender] = 1;
            nb_recus ++;
            if 
               ::valrec < mincal ->
                  mincal = valrec;
               ::valrec >= mincal ->;
            fi;
      :: /* test terminaison */
         decision == 0 && emission == 1 ->
            Test_Decision();
      :: decision == 1 && emission == 1 &&deja_emis == 0 ->
         EnvoiMinVal();         
      :: deja_emis == 1 -> break;
   od;
   printf("%d : le minimum est %d !!!\n", id, mincal);
   atomic { finished_count++; }
}

/**************************************************************/
/* Creates a network of 6 nodes 
   The structure of the network is given by array voisins */ 

init{
   byte proc; 
   
   atomic {
      proc=0;
      do
         :: proc <N ->
            run node(proc); 
            proc++
         :: proc == N -> break 
      od
   }
   do
        :: (finished_count < N) -> skip;
        :: (finished_count == N) -> break;
   od;
   printf("%d messages envoyés\n",message_count);
}
